def align_answer(frame, bits, keys):
    if frame.duplicated(screen.KEYS).any() or keys.duplicated(screen.KEYS).any():
        raise ValueError("duplicate answer keys")
    answer = frame[screen.KEYS].copy()
    answer["label"] = bits
    answer = keys.merge(answer, on=screen.KEYS, how="left", validate="one_to_one", sort=False)
    if len(answer) != len(frame) or answer.isna().any().any() or not answer.label.isin([0, 1]).all():
        raise ValueError("answer keys or binary values invalid")
    answer["label"] = answer.label.astype(np.int8)
    if not answer[screen.KEYS].equals(keys):
        raise ValueError("answer key order changed")
    return answer

canonical = SimpleNamespace(align_answer=align_answer, decoder=decoder)
